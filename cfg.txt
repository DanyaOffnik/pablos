<script>
document.addEventListener('DOMContentLoaded', function () {
  const openBtns = document.querySelectorAll('[data-role="open-search-widget"]');
  const widget = document.querySelector('[data-role="search-widget"]');
  const backdrop = widget ? widget.querySelector('.product-search-widget__backdrop') : null;
  const closeEls = document.querySelectorAll('[data-role="close-search-widget"]');
  const input = document.querySelector('[data-role="search-input"]');
  const results = document.querySelector('[data-role="search-results"]');
  const list = document.querySelector('[data-role="search-results-list"]');
  if (!openBtns.length || !widget || !backdrop || !input || !list) return;

  let isClosing = false;
  let lockedScrollY = 0;
  let searchTimer = null;
  let activeSearchController = null;

  const ajaxUrl = 'https://ecolares.ru/wp-admin/admin-ajax.php';
  const ajaxNonce = 'e882aa5410';

  function escapeHtml(value) {
    return String(value || '')
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/\"/g, '&quot;')
      .replace(/'/g, '&#039;');
  }

  function renderResults(items) {
    if (!Array.isArray(items) || !items.length) {
      list.innerHTML = '<div class="product-search-widget__empty">Ничего не найдено</div>';
      return;
    }

    list.innerHTML = items.map(function (item) {
      let metaLine = '';
      if (item.sku) {
        let metaHtml = '<span class="product-search-widget__item-meta-article">Артикул: ' + escapeHtml(item.sku) + '</span>';
        if (item.size_display) {
          metaHtml += '<span class="product-search-widget__item-meta-size">Размер: ' + escapeHtml(item.size_display) + '</span>';
        }
        metaLine = '<div class="product-search-widget__item-meta">' + metaHtml + '</div>';
      }
      const priceText = item.price_display || item.price_from || '';
      const priceLine = priceText ? '<div class="product-search-widget__item-price">' + escapeHtml(priceText) + '</div>' : '';
      const image = item.image_url
        ? '<img src="' + escapeHtml(item.image_url) + '" alt="' + escapeHtml(item.title) + '" loading="lazy">'
        : '';

      return ''
        + '<a href="' + escapeHtml(item.link || '#') + '" class="product-search-widget__item">'
        +   '<div class="product-search-widget__item-image">' + image + '</div>'
        +   '<div class="product-search-widget__item-body">'
        +     '<div class="product-search-widget__item-title">' + escapeHtml(item.title || '') + '</div>'
        +     metaLine
        +   '</div>'
        +   priceLine
        + '</a>';
    }).join('');
  }

  function lockPageScroll() {
    lockedScrollY = window.scrollY || window.pageYOffset || 0;
    document.body.classList.add('search-widget-open');
    document.body.style.top = '-' + lockedScrollY + 'px';
  }

  function unlockPageScroll() {
    document.body.classList.remove('search-widget-open');
    document.body.style.top = '';
    window.scrollTo(0, lockedScrollY);
  }

  function updateResultsVisibility() {
    if (!results || !list) return;
    const hasQuery = input.value.trim().length > 0;
    results.hidden = !hasQuery;
    results.classList.toggle('is-visible', hasQuery);
    if (!hasQuery) {
      list.innerHTML = '';
    }
  }

  function resetSearchState() {
    input.value = '';
    if (searchTimer) {
      clearTimeout(searchTimer);
      searchTimer = null;
    }
    if (activeSearchController) {
      activeSearchController.abort();
      activeSearchController = null;
    }
    updateResultsVisibility();
  }

  function runSearch(query) {
    if (!query) return;

    if (activeSearchController) {
      activeSearchController.abort();
    }
    activeSearchController = new AbortController();

    const params = new URLSearchParams({
      action: 'eco_product_search',
      nonce: ajaxNonce,
      q: query
    });

    fetch(ajaxUrl + '?' + params.toString(), { signal: activeSearchController.signal })
      .then(function (res) { return res.json(); })
      .then(function (payload) {
        if (!payload || !payload.success) {
          renderResults([]);
          return;
        }
        renderResults(payload.data && payload.data.items ? payload.data.items : []);
      })
      .catch(function (err) {
        if (err && err.name === 'AbortError') return;
        renderResults([]);
      })
      .finally(function () {
        activeSearchController = null;
      });
  }

  function openWidget() {
    if (widget.classList.contains('is-open') || isClosing) return;

    widget.classList.remove('is-closing');
    widget.setAttribute('aria-hidden', 'false');
    lockPageScroll();
    requestAnimationFrame(function () {
      requestAnimationFrame(function () {
        widget.classList.add('is-open');
        input.focus();
      });
    });
  }

  function closeWidget() {
    if (!widget.classList.contains('is-open') || isClosing) return;

    isClosing = true;
    widget.classList.remove('is-open');
    widget.classList.add('is-closing');

    const finishClose = function () {
      backdrop.removeEventListener('transitionend', onTransitionEnd);
      widget.classList.remove('is-closing');
      widget.setAttribute('aria-hidden', 'true');
      unlockPageScroll();
      resetSearchState();
      isClosing = false;
    };

    const onTransitionEnd = function (e) {
      if (e.target !== backdrop) return;
      finishClose();
    };

    backdrop.addEventListener('transitionend', onTransitionEnd);

    setTimeout(function () {
      if (isClosing) finishClose();
    }, 400);
  }

  openBtns.forEach(function (btn) {
    btn.addEventListener('click', openWidget);
  });
  input.addEventListener('input', function () {
    updateResultsVisibility();
    const query = input.value.trim();
    if (searchTimer) {
      clearTimeout(searchTimer);
      searchTimer = null;
    }
    if (!query) return;
    searchTimer = setTimeout(function () {
      runSearch(query);
    }, 220);
  });
  closeEls.forEach(function (el) {
